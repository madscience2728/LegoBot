# Dependencies for running on the Pi: lego_pi/relay_server.py (FastAPI +
# uvicorn), lego_pi/media_relay.py (camera/mic capture), and
# lego_control/hub_controller.py (pylgbst + bleak), since all of this
# gets deployed together and runs in the same interpreter (see

fastapi
uvicorn
pylgbst[bleak]

# media_relay.py -- camera + mic capture, and the WebSocket library
# uvicorn needs to actually SERVE its /media endpoint. `pip install
# uvicorn` alone does NOT include WebSocket support -- without this,
# uvicorn 404s every WebSocket handshake with "No supported WebSocket
# library detected" instead of upgrading the connection, which looks
# like a media_relay.py bug but is really just a missing dependency.
opencv-python-headless
sounddevice
numpy
websockets
